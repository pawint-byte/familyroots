Detect if a URL is an AWIN merchant:

function isAwinMerchant(hostname) {
  for (const [name, merchant] of Object.entries(AWIN_MERCHANTS)) {
    if (merchant.hostnames.some(h => hostname.includes(h))) {
      return { name, id: merchant.id };
    }
  }
  return null;
}
Convert a product URL to an AWIN affiliate link:

function convertToAwinAffiliateLink(originalUrl) {
  const hostname = new URL(originalUrl).hostname.toLowerCase();
  const merchant = isAwinMerchant(hostname);
  if (!merchant) return originalUrl;
  return `https://www.awin1.com/cread.php?awinmid=${merchant.id}&awinaffid=${AWIN_PUBLISHER_ID}&ued=${encodeURIComponent(originalUrl)}`;
}
Amazon affiliate (separate from AWIN):

function addAmazonTag(url) {
  const urlObj = new URL(url);
  urlObj.searchParams.set('tag', 'pawint-20');
  return urlObj.toString();
}
Travelpayouts deeplink API (for Klook, Kiwi, etc.):

async function convertToTravelpayoutsLink(url) {
  const token = process.env.TRAVELPAYOUTS_API_TOKEN;
  const response = await fetch(
    `https://api.travelpayouts.com/v2/statistics/deeplink?token=${token}&marker=698229&url=${encodeURIComponent(url)}`
  );
  const data = await response.json();
  return data.result.links[0].partner_url;
}
